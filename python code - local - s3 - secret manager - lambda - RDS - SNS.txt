import os
import json
import csv
import io
import re
import boto3
import pymysql

from urllib.parse import unquote_plus
from botocore.config import Config


# =========================================================
# AWS CLIENTS
# =========================================================

s3 = boto3.client("s3")

secrets_manager = boto3.client("secretsmanager")

sns = boto3.client(
    "sns",
    region_name="ap-south-1",
    config=Config(
        connect_timeout=5,
        read_timeout=10,
        retries={
            "max_attempts": 2,
            "mode": "standard"
        }
    )
)


# =========================================================
# SNS TOPIC ARNS
# =========================================================

SUCCESS_TOPIC_ARN = (
    "arn:aws:sns:ap-south-1:678494329917:"
    "student-csv-processing-topic"
)

FAILURE_TOPIC_ARN = (
    "arn:aws:sns:ap-south-1:678494329917:"
    "student-csv-failure"
)


# =========================================================
# ENVIRONMENT VARIABLES
# =========================================================

DB_HOST = os.environ["DB_HOST"]
DB_PORT = int(os.environ.get("DB_PORT", "3306"))
DB_USER = os.environ["DB_USER"]
DB_SECRET_ARN = os.environ["DB_SECRET_ARN"]

DB_NAME = os.environ.get(
    "DB_NAME",
    "studentdb"
).strip()


# Validate database name

if not DB_NAME or DB_NAME.lower() == "mysql":
    DB_NAME = "studentdb"

if not re.fullmatch(r"[A-Za-z0-9_]+", DB_NAME):
    raise ValueError(
        "DB_NAME contains invalid characters."
    )


# =========================================================
# LAMBDA HANDLER
# =========================================================

def lambda_handler(event, context):

    connection = None
    current_step = 0

    bucket_name = "Unknown"
    object_key = "Unknown"

    try:

        # =================================================
        # STEP 1: S3 FILE RETRIEVAL
        # =================================================

        current_step = 1

        print(
            "STEP 1: S3 FILE RETRIEVAL - STARTED"
        )

        records = event.get("Records", [])

        if not records:
            raise ValueError(
                "No S3 event records found."
            )

        # Process the first S3 object in this event.

        s3_record = records[0]

        bucket_name = (
            s3_record["s3"]["bucket"]["name"]
        )

        object_key = unquote_plus(
            s3_record["s3"]["object"]["key"]
        )

        print(f"S3 Bucket: {bucket_name}")
        print(f"S3 Object: {object_key}")

        # Retrieve CSV from S3.

        response = s3.get_object(
            Bucket=bucket_name,
            Key=object_key
        )

        csv_content = (
            response["Body"]
            .read()
            .decode("utf-8-sig")
        )

        print("STEP 1: SUCCESS")


        # =================================================
        # STEP 2: CSV VALIDATION
        # =================================================

        current_step = 2

        print(
            "STEP 2: CSV VALIDATION - STARTED"
        )

        csv_reader = csv.DictReader(
            io.StringIO(csv_content)
        )

        required_columns = {
            "student_id",
            "name",
            "email"
        }

        headers = set(
            csv_reader.fieldnames or []
        )

        if not required_columns.issubset(headers):
            raise ValueError(
                f"Invalid CSV headers. Expected: "
                f"{sorted(required_columns)}. "
                f"Received: {csv_reader.fieldnames}"
            )

        student_records = []

        for row_number, row in enumerate(
            csv_reader,
            start=2
        ):

            student_id = (
                row.get("student_id") or ""
            ).strip()

            name = (
                row.get("name") or ""
            ).strip()

            email = (
                row.get("email") or ""
            ).strip()

            if not student_id or not name or not email:
                raise ValueError(
                    f"Missing required value "
                    f"in CSV row {row_number}."
                )

            try:
                student_id = int(student_id)

            except ValueError:
                raise ValueError(
                    f"Invalid student_id "
                    f"in CSV row {row_number}."
                )

            student_records.append(
                (student_id, name, email)
            )

        if not student_records:
            raise ValueError(
                "CSV file contains no student records."
            )

        print(
            f"Validated student records: "
            f"{len(student_records)}"
        )

        print("STEP 2: SUCCESS")


        # =================================================
        # STEP 3: SECRETS MANAGER
        # =================================================

        current_step = 3

        print(
            "STEP 3: SECRETS MANAGER - STARTED"
        )

        secret_response = (
            secrets_manager.get_secret_value(
                SecretId=DB_SECRET_ARN
            )
        )

        secret_data = json.loads(
            secret_response["SecretString"]
        )

        # Your existing secret uses the "password" key.

        db_password = secret_data["password"]

        print("STEP 3: SUCCESS")


        # =================================================
        # STEP 4: RDS CONNECTION
        # =================================================

        current_step = 4

        print(
            "STEP 4: RDS CONNECTION - STARTED"
        )

        connection = pymysql.connect(
            host=DB_HOST,
            port=DB_PORT,
            user=DB_USER,
            password=db_password,
            connect_timeout=10,
            cursorclass=pymysql.cursors.DictCursor,
            autocommit=False
        )

        print("STEP 4: SUCCESS")


        # =================================================
        # STEP 5: INSERT / UPDATE STUDENT RECORDS
        # =================================================

        current_step = 5

        print(
            "STEP 5: INSERT STUDENT RECORDS - STARTED"
        )

        with connection.cursor() as cursor:

            # Create database if it does not exist.

            cursor.execute(
                f"CREATE DATABASE IF NOT EXISTS `{DB_NAME}`"
            )

            # Select the application database.

            cursor.execute(
                f"USE `{DB_NAME}`"
            )

            # Create table if it does not exist.

            cursor.execute("""
                CREATE TABLE IF NOT EXISTS students (
                    student_id INT PRIMARY KEY,
                    name VARCHAR(100) NOT NULL,
                    email VARCHAR(255) NOT NULL
                )
            """)

            # Insert new records.
            # Update records when student_id already exists.

            insert_query = """
                INSERT INTO students (
                    student_id,
                    name,
                    email
                )
                VALUES (%s, %s, %s)
                ON DUPLICATE KEY UPDATE
                    name = VALUES(name),
                    email = VALUES(email)
            """

            cursor.executemany(
                insert_query,
                student_records
            )

        # Commit changes before verification.

        connection.commit()

        print(
            f"INSERT/UPDATE COMPLETED: "
            f"{len(student_records)} records processed."
        )

        print("STEP 5: SUCCESS")


        # =================================================
        # STEP 6: VERIFY DATA IN RDS
        # =================================================

        print(
            "RDS DATA VERIFICATION: STARTED"
        )

        with connection.cursor() as cursor:

            cursor.execute(
                f"SELECT student_id, name, email "
                f"FROM `{DB_NAME}`.`students` "
                f"ORDER BY student_id"
            )

            results = cursor.fetchall()

        print(
            "RDS DATA VERIFICATION: SUCCESS"
        )

        print(
            "========== RDS DATA VERIFICATION =========="
        )

        print(
            json.dumps(
                results,
                indent=2,
                default=str
            )
        )

        print(
            "============================================"
        )


        # =================================================
        # STEP 7: SNS SUCCESS NOTIFICATION
        # =================================================

        current_step = 7

        print(
            "STEP 7: SNS SUCCESS NOTIFICATION - STARTED"
        )

        try:

            sns_response = sns.publish(
                TopicArn=SUCCESS_TOPIC_ARN,
                Subject=(
                    "Student CSV Processing - SUCCESS"
                ),
                Message=(
                    "Student CSV processing "
                    "completed successfully.\n\n"
                    f"S3 Bucket: {bucket_name}\n"
                    f"S3 File: {object_key}\n"
                    f"Database: {DB_NAME}\n"
                    "Table: students\n"
                    f"Records processed: "
                    f"{len(student_records)}\n"
                    f"Records found in RDS: "
                    f"{len(results)}\n"
                    "RDS verification: SUCCESS\n"
                )
            )

            print(
                "SNS SUCCESS NOTIFICATION SENT"
            )

            print(
                f"SNS Message ID: "
                f"{sns_response['MessageId']}"
            )

        except Exception as sns_error:

            print(
                "SNS SUCCESS NOTIFICATION FAILED: "
                f"{str(sns_error)}"
            )

            # Let the outer exception handler report
            # the failure and attempt a failure notification.

            raise


        # =================================================
        # FINAL SUCCESS RESPONSE
        # =================================================

        return {
            "statusCode": 200,
            "body": json.dumps({
                "message": (
                    "Student CSV processed and "
                    "RDS data verified."
                ),
                "bucket": bucket_name,
                "file": object_key,
                "database": DB_NAME,
                "table": "students",
                "records_processed": len(
                    student_records
                ),
                "records_found_in_rds": len(
                    results
                ),
                "verified_records": results
            }, default=str)
        }


    # =====================================================
    # ERROR HANDLING
    # =====================================================

    except Exception as error:

        if connection:
            try:
                connection.rollback()

            except Exception as rollback_error:
                print(
                    f"RDS ROLLBACK FAILED: "
                    f"{rollback_error}"
                )

        print(
            f"STEP {current_step}: FAILED"
        )

        print(
            f"ERROR: {str(error)}"
        )


        # =================================================
        # SNS FAILURE NOTIFICATION
        # =================================================

        print(
            "SNS FAILURE NOTIFICATION - STARTED"
        )

        try:

            failure_response = sns.publish(
                TopicArn=FAILURE_TOPIC_ARN,
                Subject=(
                    "FAILURE: Student CSV Processing"
                ),
                Message=json.dumps({
                    "status": "FAILURE",
                    "message": (
                        "Student CSV processing failed."
                    ),
                    "failed_step": current_step,
                    "error": str(error),
                    "bucket": bucket_name,
                    "file": object_key
                }, indent=2, default=str)
            )

            print(
                "SNS FAILURE NOTIFICATION SENT"
            )

            print(
                f"SNS Failure Message ID: "
                f"{failure_response['MessageId']}"
            )

        except Exception as sns_error:

            print(
                "SNS FAILURE NOTIFICATION FAILED: "
                f"{str(sns_error)}"
            )


        # Report the original Lambda execution failure.

        raise


    # =====================================================
    # CLEANUP
    # =====================================================

    finally:

        if connection:

            connection.close()

            print(
                "RDS connection closed."
            )
